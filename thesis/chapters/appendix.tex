\chapter{Reproduction Commands}

\section{Single Star-LoRA Run}

\begin{lstlisting}[language=bash]
python train.py \
  --model_name Qwen/Qwen2.5-0.5B \
  --dataset_name glue \
  --dataset_config sst2 \
  --text_column sentence \
  --label_column label \
  --num_samples 5000 \
  --method dataset_aware_adalora \
  --base_rank 8 \
  --torch_dtype float32 \
  --epochs 3 \
  --batch_size 8 \
  --lr 1e-4 \
  --seed 13 \
  --output_dir outputs/reproduction/star_n5000_seed13
\end{lstlisting}

\section{Multi-Seed Sweep}

\begin{lstlisting}[language=bash]
python run_experiments.py \
  --methods full lora adalora dataset_aware_adalora \
  --sample_budgets 100 500 1000 5000 10000 \
  --seeds 13 21 42 87 100 \
  --torch_dtype float32 \
  --epochs 3 \
  --batch_size 8 \
  --lr 1e-4 \
  --output_root outputs/reproduction_sweep
\end{lstlisting}

\section{Artifact Interpretation}

\texttt{results.csv} contains one row per method, budget, and seed.
\texttt{summary.csv} contains aggregate means and standard deviations.
\texttt{experiment\_info.json} contains selected settings and metrics.
\texttt{stability\_scores.csv} contains per-step importance logs. Results from different
\texttt{output\_root} directories should not be pooled unless all experimental settings match.
